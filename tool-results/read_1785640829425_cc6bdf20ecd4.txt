   387→// ===== Ejecutores =====
   388→async function ejecutarTool(name: string, args: any): Promise<{ success: boolean; data: any; mensaje: string }> {
   389→  try {
   390→    switch (name) {
   391→      case 'crear_tarea': {
   392→        const tarea = await db.tarea.create({
   393→          data: {
   394→            titulo: String(args.titulo).slice(0, 200),
   395→            descripcion: args.descripcion ?? null,
   396→            prioridad: args.prioridad ?? 'media',
   397→            categoria: args.categoria ?? null,
   398→            fechaLimite: args.fechaLimite ? new Date(args.fechaLimite) : null,
   399→            origen: 'voz',
   400→          },
   401→        });
   402→        return {
   403→          success: true,
   404→          data: tarea,
   405→          mensaje: `✅ Tarea creada: "${tarea.titulo}" (prioridad: ${tarea.prioridad})`,
   406→        };
   407→      }
   408→      case 'listar_tareas': {
   409→        const estado = args.estado === 'todas' || !args.estado ? undefined : args.estado;
   410→        const tareas = await db.tarea.findMany({
   411→          where: estado ? { estado } : { estado: { not: 'cancelada' } },
   412→          orderBy: [{ prioridad: 'desc' }, { createdAt: 'desc' }],
   413→          take: 50,
   414→        });
   415→        return {
   416→          success: true,
   417→          data: tareas,
   418→          mensaje: `📋 ${tareas.length} tarea(s) ${estado ? `con estado "${estado}"` : 'activas'}`,
   419→        };
   420→      }
   421→      case 'completar_tarea': {
   422→        let tarea;
   423→        if (args.id) {
   424→          tarea = await db.tarea.update({
   425→            where: { id: args.id },
   426→            data: { estado: 'completada', completadaEn: new Date() },
   427→          });
   428→        } else if (args.buscar) {
   429→          const encontrada = await db.tarea.findFirst({
   430→            where: { titulo: { contains: args.buscar }, estado: 'pendiente' },
   431→            orderBy: { createdAt: 'desc' },
   432→          });
   433→          if (!encontrada) {
   434→            return { success: false, data: null, mensaje: `No encontré tarea pendiente con "${args.buscar}"` };
   435→          }
   436→          tarea = await db.tarea.update({
   437→            where: { id: encontrada.id },
   438→            data: { estado: 'completada', completadaEn: new Date() },
   439→          });
   440→        }
   441→        if (!tarea) return { success: false, data: null, mensaje: 'Falta id o buscar' };
   442→        return { success: true, data: tarea, mensaje: `✅ Tarea completada: "${tarea.titulo}"` };
   443→      }
   444→      case 'eliminar_tarea': {
   445→        let tarea;
   446→        if (args.id) {
   447→          tarea = await db.tarea.delete({ where: { id: args.id } });
   448→        } else if (args.buscar) {
   449→          const encontrada = await db.tarea.findFirst({
   450→            where: { titulo: { contains: args.buscar } },
   451→            orderBy: { createdAt: 'desc' },
   452→          });
   453→          if (!encontrada) return { success: false, data: null, mensaje: `No encontré tarea con "${args.buscar}"` };
   454→          tarea = await db.tarea.delete({ where: { id: encontrada.id } });
   455→        }
   456→        if (!tarea) return { success: false, data: null, mensaje: 'Falta id o buscar' };
   457→        return { success: true, data: tarea, mensaje: `🗑️ Tarea eliminada: "${tarea.titulo}"` };
   458→      }
   459→      case 'crear_nota': {
   460→        const nota = await db.nota.create({
   461→          data: {
   462→            titulo: String(args.titulo).slice(0, 200),
   463→            contenido: String(args.contenido),
   464→            color: args.color ?? 'amarillo',
   465→            origen: 'voz',
   466→          },
   467→        });
   468→        return { success: true, data: nota, mensaje: `📝 Nota creada: "${nota.titulo}"` };
   469→      }
   470→      case 'listar_notas': {
   471→        const notas = await db.nota.findMany({
   472→          where: { archivada: false },
   473→          orderBy: [{ fijada: 'desc' }, { createdAt: 'desc' }],
   474→          take: 50,
   475→        });
   476→        return { success: true, data: notas, mensaje: `📝 ${notas.length} nota(s)` };
   477→      }
   478→      case 'crear_recordatorio': {
   479→        const rec = await db.recordatorio.create({
   480→          data: {
   481→            titulo: String(args.titulo).slice(0, 200),
   482→            descripcion: args.descripcion ?? null,
   483→            fechaHora: new Date(args.fechaHora),
   484→            recurrencia: args.recurrencia ?? 'unica',
   485→            origen: 'voz',
   486→          },
   487→        });
   488→        return {
   489→          success: true,
   490→          data: rec,
   491→          mensaje: `⏰ Recordatorio: "${rec.titulo}" para ${rec.fechaHora.toLocaleString('es-MX')}`,
   492→        };
   493→      }
   494→      case 'listar_recordatorios': {
   495→        const ahora = new Date();
   496→        const recs = await db.recordatorio.findMany({
   497→          where: { fechaHora: { gte: ahora }, estado: 'pendiente' },
   498→          orderBy: { fechaHora: 'asc' },
   499→          take: 20,
   500→        });
   501→        return { success: true, data: recs, mensaje: `⏰ ${recs.length} recordatorio(s) próximos` };
   502→      }
   503→      case 'calcular': {
   504→        const resultado = evaluarExpresionSegura(args.expresion);
   505→        if (resultado === null) {
   506→          return { success: false, data: null, mensaje: 'No pude evaluar esa expresión' };
   507→        }
   508→        return { success: true, data: { expresion: args.expresion, resultado }, mensaje: `🧮 ${args.expresion} = ${resultado}` };
   509→      }
   510→      case 'resumir_conversacion': {
   511→        const [tareas, notas, recs] = await Promise.all([
   512→          db.tarea.count({ where: { estado: 'pendiente' } }),
   513→          db.nota.count({ where: { archivada: false } }),
   514→          db.recordatorio.count({ where: { estado: 'pendiente', fechaHora: { gte: new Date() } } }),
   515→        ]);
   516→        const proximasTareas = await db.tarea.findMany({
   517→          where: { estado: 'pendiente' },
   518→          orderBy: [{ prioridad: 'desc' }],
   519→          take: 5,
   520→        });
   521→        return {
   522→          success: true,
   523→          data: { tareasPendientes: tareas, notasActivas: notas, recordatoriosProximos: recs, proximasTareas },
   524→          mensaje: `📊 Resumen: ${tareas} tareas pendientes, ${notas} notas, ${recs} recordatorios próximos`,
   525→        };
   526→      }
   527→      // ===== Nuevas tools =====
   528→      case 'obtener_fecha_hora': {
   529→        const ahora = new Date();
   530→        const fechaHora = ahora.toLocaleString('es-MX', {
   531→          timeZone: 'America/Mazatlan',
   532→          weekday: 'long',
   533→          year: 'numeric',
   534→          month: 'long',
   535→          day: 'numeric',
   536→          hour: '2-digit',
   537→          minute: '2-digit',
   538→          second: '2-digit',
   539→        });
   540→        return {
   541→          success: true,
   542→          data: { fechaHora, iso: ahora.toISOString() },
   543→          mensaje: `🕐 ${fechaHora}`,
   544→        };
   545→      }
   546→      case 'buscar_conocimiento': {

   901→function evaluarExpresionSegura(expr: string): number | null {
   902→  try {
   903→    let e = expr.toLowerCase().trim();
   904→    e = e.replace(/(\d+(?:\.\d+)?)\s*%\s*de\s*(\d+(?:\.\d+)?)/g, '($1/100*$2)');
   905→    e = e.replace(/por ciento/g, '/100');
   906→    e = e.replace(/por/g, '*');
   907→    e = e.replace(/más/g, '+');
   908→    e = e.replace(/menos/g, '-');
   909→    e = e.replace(/entre/g, '/');
   910→    e = e.replace(/sqrt\s*\(([^)]+)\)/g, 'Math.sqrt($1)');
   911→    e = e.replace(/raíz\s*cuadrada\s*de\s*(\d+(?:\.\d+)?)/g, 'Math.sqrt($1)');
   912→    if (!/^[\d+\-*/().\sMathsqr]+$/.test(e.replace(/Math\./g, ''))) {
   913→      return null;
   914→    }
   915→    const resultado = Function(`"use strict"; return (${e});`)();
   916→    if (typeof resultado === 'number' && !isNaN(resultado)) {
   917→      return Math.round(resultado * 1e6) / 1e6;
   918→    }
   919→    return null;
   920→  } catch {
   921→    return null;
   922→  }
