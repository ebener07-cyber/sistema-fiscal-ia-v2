    36→// ===== Parser de CSV =====
    37→function parseCSV(texto: string): MovimientoImportado[] {
    38→  const lineas = texto.split(/\r?\n/).filter(l => l.trim());
    39→  if (lineas.length === 0) return [];
    40→
    41→  // Detectar separador
    42→  const separador = lineas[0].includes(';') ? ';' : lineas[0].includes('\t') ? '\t' : ',';
    43→
    44→  // Saltar header si existe
    45→  let empezarDesde = 0;
    46→  if (lineas[0].toLowerCase().includes('fecha') || lineas[0].toLowerCase().includes('date')) {
    47→    empezarDesde = 1;
    48→  }
    49→
    50→  const movimientos: MovimientoImportado[] = [];
    51→
    52→  for (let i = empezarDesde; i < lineas.length; i++) {
    53→    const partes = lineas[i].split(separador).map(p => p.trim().replace(/"/g, ''));
    54→    if (partes.length < 3) continue;
    55→
    56→    try {
    57→      // Parsear fecha (DD/MM/YYYY o YYYY-MM-DD o DD-MM-YYYY)
    58→      let fecha: Date;
    59→      const fechaStr = partes[0];
    60→      if (fechaStr.match(/^\d{4}-\d{2}-\d{2}/)) {
    61→        fecha = new Date(fechaStr);
    62→      } else if (fechaStr.match(/^\d{1,2}\/\d{1,2}\/\d{4}/)) {
    63→        const [dia, mes, anio] = fechaStr.split('/');
    64→        fecha = new Date(parseInt(anio), parseInt(mes) - 1, parseInt(dia));
    65→      } else if (fechaStr.match(/^\d{1,2}-\d{1,2}-\d{4}/)) {
    66→        const [dia, mes, anio] = fechaStr.split('-');
    67→        fecha = new Date(parseInt(anio), parseInt(mes) - 1, parseInt(dia));
    68→      } else {
    69→        continue;
    70→      }
    71→
    72→      // Buscar monto (puede estar en columna 2, 3, 4, etc.)
    73→      let monto = 0;
    74→      let concepto = partes[1] || 'Movimiento';
    75→      for (let j = 2; j < partes.length; j++) {
    76→        const valor = partes[j].replace(/[$,\s]/g, '').replace(',', '.');
    77→        // Intentar parsear como número
    78→        const cleaned = valor.replace(/[^0-9.-]/g, '');
    79→        const parsed = parseFloat(cleaned);
    80→        if (!isNaN(parsed) && parsed !== 0) {
    81→          monto = parsed;
    82→          if (partes[j + 1]) concepto = `${concepto} ${partes[j + 1]}`.trim();
    83→          break;
    84→        }
    85→      }
    86→
    87→      if (monto === 0) continue;
    88→
    89→      movimientos.push({ fecha, concepto, monto });
    90→    } catch {
    91→      continue;
    92→    }
    93→  }
    94→
    95→  return movimientos;
    96→}

    98→// ===== Parser de Excel (.xlsx) usando exceljs =====
    99→// Soporta formatos:
   100→//   - Genérico: Fecha | Concepto | Monto (con signo)
   101→//   - Banorte: CUENTA | FECHA | REFERENCIA | DESCRIPCIÓN | DEPÓSITOS | RETIROS | SALDO
   102→//   - BBVA: Fecha | Concepto | Depósitos | Retiros | Saldo
   103→//   - Cualquier banco con columnas separadas de Cargo/Abono o Depósito/Retiro
   104→async function parseExcel(buffer: Buffer): Promise<MovimientoImportado[]> {
   105→  const ExcelJS = (await import('exceljs')).default;
   106→  const wb = new ExcelJS.Workbook();
   107→  await wb.xlsx.load(buffer);
   108→
   109→  const movimientos: MovimientoImportado[] = [];
   110→
   111→  for (const ws of wb.worksheets) {
   112→    // Buscar fila de headers (puede no ser la primera)
   113→    let headerRow = 1;
   114→    const headers: string[] = [];
   115→    for (let r = 1; r <= Math.min(10, ws.rowCount); r++) {
   116→      const fila = ws.getRow(r);
   117→      const tempHeaders: string[] = [];
   118→      fila.eachCell((cell, col) => {
   119→        tempHeaders[col] = String(cell.value || '').toLowerCase().trim();
   120→      });
   121→      const joined = tempHeaders.join('|');
   122→      // Detectar fila con headers de banco
   123→      if (
   124→        joined.includes('fecha') &&
   125→        (joined.includes('descripci') || joined.includes('concepto') || joined.includes('descrip'))
   126→      ) {
   127→        for (let c = 1; c <= tempHeaders.length; c++) headers[c] = tempHeaders[c];
   128→        headerRow = r;
   129→        break;
   130→      }
   131→    }
   132→
   133→    // Si no encontró headers, intentar con la primera fila
   134→    if (headers.length === 0) {
   135→      const primeraFila = ws.getRow(1);
   136→      primeraFila.eachCell((cell, col) => {
   137→        headers[col] = String(cell.value || '').toLowerCase().trim();
   138→      });
   139→    }
   140→
   141→    // Mapear columnas
   142→    let colFecha = 1, colConcepto = 2, colDeposito = 0, colRetiro = 0, colMonto = 0;
   143→    let colDescripcionDetallada = 0;
   144→    let colReferencia = 0;
   145→
   146→    for (let c = 1; c <= Math.max(headers.length, 20); c++) {
   147→      const h = headers[c] || '';
   148→      if (h.includes('fecha') && !h.includes('opera')) colFecha = c;
   149→      else if (h.includes('fecha')) colFecha = c; // "FECHA DE OPERACIÓN" también cuenta
   150→      if (h === 'descripción' || h === 'descripcion' || h.includes('descrip') || h.includes('concepto') || h.includes('detalle')) {
   151→        if (!colConcepto || colConcepto === 2) colConcepto = c;
   152→      }
   153→      if (h.includes('descripción detallada') || h.includes('descripcion detallada')) colDescripcionDetallada = c;
   154→      if (h.includes('referencia')) colReferencia = c;
   155→      // Depósitos / Abonos / Créditos
   156→      if (h.includes('depósito') || h.includes('deposito') || h.includes('abono') || h.includes('crédito') || h.includes('credito') || h.includes('ingreso')) {
   157→        colDeposito = c;
   158→      }
   159→      // Retiros / Cargos / Débitos
   160→      if (h.includes('retiro') || h.includes('cargo') || h.includes('débito') || h.includes('debito') || h.includes('egreso')) {
   161→        colRetiro = c;
   162→      }
   163→      // Monto único (con signo)
   164→      if (h.includes('monto') || h.includes('importe') || h.includes('amount') || h.includes('movimiento')) {
   165→        colMonto = c;
   166→      }
   167→    }
   168→
   169→    const filaInicio = headerRow + 1;
   170→
   171→    for (let r = filaInicio; r <= ws.rowCount; r++) {
   172→      const fila = ws.getRow(r);
   173→      try {
   174→        const cellFecha = fila.getCell(colFecha).value;
   175→
   176→        // Saltar filas vacías
   177→        if (!cellFecha) continue;
   178→
   179→        // Parsear fecha
   180→        let fecha: Date | null = null;
   181→        if (cellFecha instanceof Date) {
   182→          fecha = cellFecha;
   183→        } else if (typeof cellFecha === 'number') {
   184→          // Excel serial date
   185→          fecha = new Date(Date.UTC(1899, 11, 30) + cellFecha * 24 * 60 * 60 * 1000);
   186→        } else if (typeof cellFecha === 'string') {
   187→          if (cellFecha.match(/^\d{4}-\d{2}-\d{2}/)) {
   188→            fecha = new Date(cellFecha);
   189→          } else if (cellFecha.match(/^\d{1,2}\/\d{1,2}\/\d{4}/)) {
   190→            const [dia, mes, anio] = cellFecha.split('/');
   191→            fecha = new Date(parseInt(anio), parseInt(mes) - 1, parseInt(dia));
   192→          } else if (cellFecha.match(/^\d{1,2}-\d{1,2}-\d{4}/)) {
   193→            const [dia, mes, anio] = cellFecha.split('-');
   194→            fecha = new Date(parseInt(anio), parseInt(mes) - 1, parseInt(dia));
   195→          }
   196→        }
   197→        if (!fecha || isNaN(fecha.getTime())) continue;
   198→
   199→        // Concepto (principal + descripción detallada si existe)
   200→        const conceptoBase = String(fila.getCell(colConcepto).value || 'Movimiento').trim();
   201→        let concepto = conceptoBase;
   202→        if (colDescripcionDetallada) {
   203→          const detalle = String(fila.getCell(colDescripcionDetallada).value || '').trim();
   204→          if (detalle && detalle !== '-' && detalle !== conceptoBase) {
   205→            concepto = `${conceptoBase} — ${detalle}`.slice(0, 500);
   206→          }
   207→        }
   208→        if (colReferencia) {
   209→          const ref = String(fila.getCell(colReferencia).value || '').trim();
   210→          if (ref && ref !== '-') {
   211→            concepto = `Ref: ${ref} · ${concepto}`.slice(0, 500);
   212→          }
   213→        }
   214→
   215→        // Calcular monto según el formato detectado
   216→        let monto = 0;
   217→
   218→        // Caso 1: Banorte-style — columnas separadas Depósito/Retiro
   219→        if (colDeposito || colRetiro) {
   220→          let deposito = 0, retiro = 0;
   221→          if (colDeposito) {
   222→            const val = parseNumberFromCell(fila.getCell(colDeposito).value);
   223→            deposito = val;
   224→          }
   225→          if (colRetiro) {
   226→            const val = parseNumberFromCell(fila.getCell(colRetiro).value);
   227→            retiro = val;
   228→          }
   229→          monto = deposito - retiro;
   230→        }
   231→        // Caso 2: Monto único con signo
   232→        else if (colMonto) {
   233→          monto = parseNumberFromCell(fila.getCell(colMonto).value);
   234→        }
   235→        // Caso 3: Fallback — buscar cualquier número en la fila después de la columna concepto
   236→        else {
   237→          for (let c = colConcepto + 1; c <= Math.min(fila.cellCount, 15); c++) {
   238→            const val = parseNumberFromCell(fila.getCell(c).value);
   239→            if (val !== 0) {
   240→              monto = val;
   241→              break;
   242→            }
   243→          }
   244→        }
   245→
   246→        if (monto === 0) continue;
   247→
   248→        movimientos.push({ fecha, concepto, monto });
   249→      } catch {
   250→        continue;
   251→      }
   252→    }
   253→  }
   254→
   255→  return movimientos;
   256→}
   257→
   258→function parseNumberFromCell(value: any): number {
   259→  if (value === null || value === undefined || value === '' || value === '-') return 0;
   260→  if (typeof value === 'number') return value;
   261→  if (typeof value === 'string') {
   262→    // Limpiar: quitar $, comas, espacios, signos de moneda
   263→    const cleaned = value.replace(/[$,\s]/g, '').replace(/[^0-9.-]/g, '');
   264→    return parseFloat(cleaned) || 0;
   265→  }
   266→  if (typeof value === 'object' && 'result' in value) {
   267→    return parseFloat(String(value.result)) || 0;
   268→  }
   269→  return 0;
   270→}

   272→// ===== Parser de texto extraído de PDF =====
   273→
   274→/**
   275→ * Extrae texto de un PDF sin usar librerías externas (compatible con Vercel serverless).
   276→ * 
   277→ * Funciona buscando texto en los content streams del PDF:
   278→ * 1. Busca streams entre "stream" y "endstream"
   279→ * 2. Intenta descomprimir con zlib si están comprimidos con FlateDecode
   280→ * 3. Extrae texto de operadores Tj y TJ dentro de bloques BT/ET
   281→ */
   282→function extraerTextoPDF(buffer: Buffer): string {
   283→  const zlib = require('zlib');
   284→  const textoCompleto: string[] = [];
   285→
   286→  // Convertir buffer a string latin1 para preservar bytes
   287→  const pdfStr = buffer.toString('latin1');
   288→
   289→  // Buscar todos los streams en el PDF
   290→  const streamRegex = /stream\r?\n([\s\S]*?)endstream/g;
   291→  let streamMatch;
   292→
   293→  while ((streamMatch = streamRegex.exec(pdfStr)) !== null) {
   294→    const streamData = streamMatch[1];
   295→
   296→    try {
   297→      // Intentar descomprimir con FlateDecode (zlib)
   298→      let decompressed: string;
   299→      try {
   300→        const compressed = Buffer.from(streamData, 'latin1');
   301→        const decompressedBuffer = zlib.inflateSync(compressed);
   302→        decompressed = decompressedBuffer.toString('latin1');
   303→      } catch {
   304→        // Si no se puede descomprimir, usar el texto tal cual
   305→        decompressed = streamData;
   306→      }
   307→
   308→      // Extraer texto de operadores Tj: (texto) Tj
   309→      const tjRegex = /\(([^)]{1,200})\)\s*Tj/g;
   310→      let tjMatch;
   311→      while ((tjMatch = tjRegex.exec(decompressed)) !== null) {
   312→        const text = decodePdfString(tjMatch[1]);
   313→        if (text && text.trim()) {
   314→          textoCompleto.push(text);
   315→        }
   316→      }
   317→
   318→      // Extraer texto de operadores TJ: [(texto1) -250 (texto2)] TJ
   319→      const tjArrayRegex = /\[([^\]]{1,500})\]\s*TJ/g;
   320→      let tjArrayMatch;
   321→      while ((tjArrayMatch = tjArrayRegex.exec(decompressed)) !== null) {
   322→        const arrayContent = tjArrayMatch[1];
   323→        // Extraer todos los strings entre paréntesis
   324→        const stringParts: string[] = [];
   325→        const partRegex = /\(([^)]{1,200})\)/g;
   326→        let partMatch;
   327→        while ((partMatch = partRegex.exec(arrayContent)) !== null) {
   328→          stringParts.push(decodePdfString(partMatch[1]));
   329→        }
   330→        if (stringParts.length > 0) {
   331→          textoCompleto.push(stringParts.join(''));
   332→        }
   333→      }
   334→    } catch {
   335→      // Saltar streams que no se pueden procesar
   336→      continue;
   337→    }
   338→  }
   339→
   340→  // También buscar texto que no esté en streams (algunos PDFs simples)
   341→  const simpleTextRegex = /\(([\w\s\/\-\.,$:#áéíóúñÁÉÍÓÚÑ]{3,80})\)\s*Tj/g;
   342→  let simpleMatch;
   343→  while ((simpleMatch = simpleTextRegex.exec(pdfStr)) !== null) {
   344→    const text = simpleMatch[1];
   345→    if (text && text.trim().length > 2) {
   346→      textoCompleto.push(text);
   347→    }
   348→  }
   349→
   350→  return textoCompleto.join('\n');
   351→}

   353→/**
   354→ * Decodifica un string de PDF (maneja escapes básicos)
   355→ */
   356→function decodePdfString(str: string): string {
   357→  return str
   358→    .replace(/\\n/g, '\n')
   359→    .replace(/\\r/g, '\r')
   360→    .replace(/\\t/g, '\t')
   361→    .replace(/\\\(/g, '(')
   362→    .replace(/\\\)/g, ')')
   363→    .replace(/\\\\/g, '\\')
   364→    .trim();
   365→}

   371→function parsePDFTexto(texto: string): MovimientoImportado[] {
   372→  // Llamar a la versión nueva que detecta múltiples cuentas

   377→/**
   378→ * Versión mejorada del parser que detecta MÚLTIPLES cuentas en el mismo PDF.
   379→ * 
   380→ * SOPORTA:
   381→ * 1. Banorte: 2 secciones (ENLACE NEGOCIOS AVANZADA + INVERSION ENLACE NEGOCIOS)

   383→ * 
   384→ * MEJORAS v2.4:
   385→ * - Detección explícita de secciones por headers (no por contexto difuso)
   386→ * - Extracción ESTRICTA de montos (NN,NNN.NN con 2 decimales obligatorios)
   387→ * - No confunde folios (7113421), CLABEs (014180655090853560), ni referencias con montos
   388→ * - Captura correctamente "SALDO FINAL DEL PERIODO ANTERIOR" como saldo inicial
   389→ * - Maneja movimientos multi-línea (descripción + varias líneas + monto+saldo)
   390→ * - Detecta fin de sección por línea "TOTAL" o "SALDO FINAL DEL PERIODO"
   391→ */
   392→function parsePDFTextoMultiCuenta(texto: string): {
   393→  movimientos: MovimientoImportado[];
   394→  seccionesDetectadas: Array<{ tipo: string; cuentaNumero: string; count: number; saldoInicial: number | null }>;
   395→} {
   396→  const lineas = texto.split(/\r?\n/);
   397→
   398→  // Mapeo de meses abreviados en español (formato Banorte: 08-ENE-26, Santander: 04-MAY-2026)
   399→  const MESES_ES: Record<string, number> = {
   400→    'ENE': 0, 'FEB': 1, 'MAR': 2, 'ABR': 3, 'MAY': 4, 'JUN': 5,
   401→    'JUL': 6, 'AGO': 7, 'SEP': 8, 'OCT': 9, 'NOV': 10, 'DIC': 11,
   402→    'ENERO': 0, 'FEBRERO': 1, 'MARZO': 2, 'ABRIL': 3, 'MAYO': 4, 'JUNIO': 5,
   403→    'JULIO': 6, 'AGOSTO': 7, 'SEPTIEMBRE': 8, 'OCTUBRE': 9, 'NOVIEMBRE': 10, 'DICIEMBRE': 11,
   404→  };
   405→
   406→  // Patrones de fecha
   407→  const patronFechaBanorte = /(\d{1,2})-([A-Z]{3,9})-(\d{2,4})/;
   408→  const patronFechaNum = /(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})/;
   409→
   410→  // ===== EXTRACCIÓN ESTRICTA DE MONTOS =====
   411→  // Solo captura números con EXACTAMENTE 2 decimales (NN,NNN.NN o NNN.NN)
   412→  // No captura números pegados a letras ni números sin decimales
   413→  function extraerMontosLinea(linea: string): number[] {
   414→    const montos: number[] = [];
   415→    const regex = /(?:^|\s)(\d{1,3}(?:,\d{3})*\.\d{2})(?:\s|$)/g;
   416→    let match;
   417→    while ((match = regex.exec(linea)) !== null) {
   418→      const valor = parseFloat(match[1].replace(/,/g, ''));
   419→      if (!isNaN(valor) && valor > 0.5) {
   420→        montos.push(valor);
   421→      }
   422→    }
   423→    return montos;
   424→  }

   426→  function quitarMontos(texto: string): string {
   427→    return texto.replace(/(?:^|\s)\d{1,3}(?:,\d{3})*\.\d{2}(?:\s|$)/g, ' ').replace(/\s+/g, ' ').trim();
   428→  }

   430→  // ===== FASE 1: DETECTAR SECCIONES =====
   431→  // Buscar headers de secciones conocidos:
   432→  // - Santander: "Detalle de movimientos cuenta de cheques." + "Detalles de movimientos Dinero Creciente"
   433→  // - Banorte: "ENLACE NEGOCIOS AVANZADA" + "INVERSION ENLACE NEGOCIOS"
   434→  interface Seccion {
   435→    tipo: 'operaciones' | 'inversion';
   436→    cuentaNum: string;
   437→    lineaInicio: number;
   438→    lineaFin: number | null;
   439→    saldoInicial: number | null;
   440→    banco: 'santander' | 'banorte' | 'desconocido';
   441→  }

   443→  const secciones: Seccion[] = [];
   444→  let seccionActual: Seccion | null = null;
   445→
   446→  function cerrarSeccion(lineaFin: number) {
   447→    if (seccionActual) {
   448→      seccionActual.lineaFin = lineaFin;
   449→      secciones.push(seccionActual);
   450→      seccionActual = null;
   451→    }
   452→  }

   454→  function iniciarSeccion(tipo: 'operaciones' | 'inversion', banco: 'santander' | 'banorte', lineaInicio: number) {
   455→    // Buscar número de cuenta en las siguientes 3 líneas
   456→    let cuentaNum = '';
   457→    for (let j = lineaInicio; j < Math.min(lineaInicio + 4, lineas.length); j++) {
   458→      // Santander: 65-50908535-6 (DD-NNNNNNNN-D)
   459→      let match = lineas[j].match(/(\d{2}-\d{8}-\d)/);
   460→      if (match) { cuentaNum = match[1]; break; }
   461→      // Banorte: 1282396470 (10 dígitos)
   462→      match = lineas[j].match(/No\.?\s*de\s*Cuenta:?\s*(\d{6,})/i);
   463→      if (match) { cuentaNum = match[1]; break; }
   464→    }
   465→    
   466→    // Buscar saldo inicial
   467→    let saldoInicial: number | null = null;
   468→    for (let j = lineaInicio; j < Math.min(lineaInicio + 6, lineas.length); j++) {
   469→      // "SALDO FINAL DEL PERIODO ANTERIOR:   $82,075.16" (Santander)
   470→      const matchSaldo = lineas[j].match(/SALDO.*?ANTERIOR:?\s*\$?([\d,]+\.\d{2})/i);
   471→      if (matchSaldo) {
   472→        saldoInicial = parseFloat(matchSaldo[1].replace(/[$,]/g, ''));
   473→        break;
   474→      }
   475→      // "SALDO ANTERIOR: $119,827.38" (Banorte)
   476→      const matchSaldoBn = lineas[j].match(/SALDO\s+ANTERIOR\s*:?\s*\$?([\d,]+\.\d{2})/i);
   477→      if (matchSaldoBn) {
   478→        saldoInicial = parseFloat(matchSaldoBn[1].replace(/[$,]/g, ''));
   479→        break;
   480→      }
   481→    }
   482→    
   483→    seccionActual = { tipo, cuentaNum, lineaInicio, lineaFin: null, saldoInicial, banco };
   484→  }
   485→
   486→  for (let i = 0; i < lineas.length; i++) {
   487→    const linea = lineas[i];
   488→    const lineaUpper = linea.toUpperCase();
   489→    
   490→    // ===== DETECCIÓN SANTANDER =====
   491→    // "Detalle de movimientos cuenta de cheques."
   492→    if (lineaUpper.includes('DETALLE DE MOVIMIENTOS CUENTA DE CHEQUES') ||
   493→        (lineaUpper.includes('DETALLE DE MOVIMIENTOS') && !lineaUpper.includes('DINERO'))) {
   494→      cerrarSeccion(i - 1);
   495→      iniciarSeccion('operaciones', 'santander', i);
   496→    }
   497→    // "Detalles de movimientos Dinero Creciente Santander." (sección de inversión)
   498→    if (lineaUpper.includes('DETALLES DE MOVIMIENTOS DINERO') || 
   499→        (lineaUpper.includes('DETALLE DE MOVIMIENTOS') && lineaUpper.includes('DINERO'))) {
   500→      cerrarSeccion(i - 1);
   501→      iniciarSeccion('inversion', 'santander', i);
   502→    }
   503→    
   504→    // ===== DETECCIÓN BANORTE =====
   505→    if (lineaUpper.includes('INVERSION') && lineaUpper.includes('ENLACE NEGOCIOS')) {
   506→      cerrarSeccion(i - 1);
   507→      iniciarSeccion('inversion', 'banorte', i);
   508→    }
   509→    if (lineaUpper.includes('ENLACE NEGOCIOS AVANZADA') || 
   510→        (lineaUpper.includes('ENLACE NEGOCIOS') && !lineaUpper.includes('INVERSION'))) {
   511→      cerrarSeccion(i - 1);
   512→      iniciarSeccion('operaciones', 'banorte', i);
   513→    }
   514→    
   515→    // ===== FIN DE SECCIÓN =====
   516→    // "TOTAL   797,935.76   371,869.72" o "SALDO FINAL DEL PERIODO:"
   517→    if (seccionActual && seccionActual.lineaFin === null) {
   518→      if (lineaUpper.match(/^TOTAL\s+\d/) || 
   519→          (lineaUpper.includes('SALDO FINAL DEL PERIODO') && !lineaUpper.includes('ANTERIOR')) ||
   520→          lineaUpper.includes('INFORMACION FISCAL') ||
   521→          lineaUpper.includes('INFORMACIÓN FISCAL')) {
   522→        cerrarSeccion(i);
   523→      }
   524→    }
   525→  }
   526→  // Cerrar última sección
   527→  cerrarSeccion(lineas.length - 1);
   528→
   529→  // ===== FASE 2: PARSEAR MOVIMIENTOS DE CADA SECCIÓN =====
   530→  function parsearSeccion(seccion: Seccion): MovimientoImportado[] {
   531→    const movimientos: MovimientoImportado[] = [];
   532→    let saldoAnterior = seccion.saldoInicial;
   533→    
   534→    let i = seccion.lineaInicio;
   535→    while (i <= (seccion.lineaFin ?? lineas.length - 1)) {
   536→      const linea = lineas[i];
   537→      if (!linea || !linea.trim()) { i++; continue; }
   538→      
   539→      // Buscar fecha al inicio de la línea (puede ser Banorte DD-ENE-26 o Santander DD-MAY-2026)
   540→      const matchFecha = linea.match(patronFechaBanorte) || linea.match(patronFechaNum);
   541→      if (!matchFecha) { i++; continue; }
   542→      
   543→      // Saltar headers
   544→      const lineaUpper = linea.toUpperCase();
   545→      if (lineaUpper.includes('FECHA') && lineaUpper.includes('FOLIO')) { i++; continue; }
   546→      if (lineaUpper.match(/^TOTAL\s/) || lineaUpper.includes('SALDO FINAL DEL PERIODO')) { i++; continue; }
   547→      
   548→      const dia = parseInt(matchFecha[1]);
   549→      let mes: number | undefined;
   550→      let mesStr = '';
   551→      
   552→      if (matchFecha[2] && isNaN(parseInt(matchFecha[2]))) {
   553→        // Es mes en texto (ENE, FEB, etc.)
   554→        mesStr = matchFecha[2].toUpperCase();
   555→        mes = MESES_ES[mesStr];
   556→      } else {
   557→        // Es mes numérico (01, 02, etc.)
   558→        mes = parseInt(matchFecha[2]) - 1;
   559→      }
   560→      
   561→      let anio = parseInt(matchFecha[3]);
   562→      if (anio < 100) anio = anio < 30 ? 2000 + anio : 1900 + anio;
   563→      
   564→      if (mes === undefined || mes < 0 || mes > 11 || dia < 1 || dia > 31) { i++; continue; }
   565→      
   566→      const fecha = new Date(anio, mes, dia, 12, 0, 0);
   567→      const restoLinea = linea.substring(matchFecha.index! + matchFecha[0].length).trim();
   568→      
   569→      // Acumular descripción y buscar monto+saldo
   570→      let concepto = restoLinea;
   571→      let montoEncontrado: number | null = null;
   572→      let saldoEncontrado: number | null = null;
   573→      let lineaMontoIdx = -1;
   574→      
   575→      // Buscar montos en la línea actual (caso simple: fecha + descripción + monto + saldo en una línea)
   576→      // Pero tambien considerar que el saldo puede ser 0.00 (que el filtro > 0.5 descarta)
   577→      // Por eso, extraemos TODOS los números con formato monetario (incluso 0.00)
   578→      function extraerMontosLineaInclusivo(linea: string): number[] {
   579→        const montos: number[] = [];
   580→        const regex = /(?:^|\s)(\d{1,3}(?:,\d{3})*\.\d{2})(?:\s|$)/g;
   581→        let match;
   582→        while ((match = regex.exec(linea)) !== null) {
   583→          const valor = parseFloat(match[1].replace(/,/g, ''));
   584→          if (!isNaN(valor)) {
   585→            montos.push(valor);
   586→          }
   587→        }
   588→        return montos;
   589→      }
   590→      
   591→      const montosAqui = extraerMontosLinea(linea);
   592→      const montosAquiInclusivo = extraerMontosLineaInclusivo(linea);
   593→      
   594→      if (montosAquiInclusivo.length >= 2) {
   595→        // Hay al menos 2 montos: el primero es el monto del movimiento, el segundo es el saldo
   596→        montoEncontrado = montosAquiInclusivo[0];
   597→        saldoEncontrado = montosAquiInclusivo[1];
   598→        // Solo aceptar si el monto es > 0.5 (el saldo puede ser 0)
   599→        if (Math.abs(montoEncontrado) < 0.5) {
   600→          montoEncontrado = null;
   601→          saldoEncontrado = null;
   602→        } else {
   603→          concepto = quitarMontos(restoLinea);
   604→          lineaMontoIdx = i;
   605→        }
   606→      } else if (montosAqui.length === 1 && restoLinea.length < 50) {
   607→        // Solo 1 monto y descripción corta — puede ser movimiento simple sin saldo
   608→        montoEncontrado = montosAqui[0];
   609→        concepto = quitarMontos(restoLinea);
   610→        lineaMontoIdx = i;
   611→      }
   612→      
   613→      // Si no, buscar en líneas siguientes (movimientos multi-línea)
   614→      if (montoEncontrado === null) {
   615→        let j = i + 1;
   616→        const limiteBusqueda = Math.min((seccion.lineaFin ?? lineas.length - 1) + 1, i + 20);
   617→        while (j < limiteBusqueda) {
   618→          const lineaSiguiente = lineas[j];
   619→          if (!lineaSiguiente || !lineaSiguiente.trim()) { j++; continue; }
   620→          
   621→          // Si encontramos otra fecha al inicio, ya no hay montos
   622→          const matchFechaSgte = lineaSiguiente.match(patronFechaBanorte) || lineaSiguiente.match(patronFechaNum);
   623→          if (matchFechaSgte && matchFechaSgte.index === 0) break;
   624→          
   625→          const lUpper = lineaSiguiente.toUpperCase();
   626→          if (lUpper.match(/^TOTAL\s/) || lUpper.includes('SALDO FINAL DEL PERIODO')) break;
   627→          
   628→          const montosSgte = extraerMontosLinea(lineaSiguiente);
   629→          if (montosSgte.length >= 2) {
   630→            montoEncontrado = montosSgte[0];
   631→            saldoEncontrado = montosSgte[1];
   632→            lineaMontoIdx = j;
   633→            break;
   634→          } else if (montosSgte.length === 1 && j > i + 1) {
   635→            montoEncontrado = montosSgte[0];
   636→            saldoEncontrado = null;
   637→            lineaMontoIdx = j;
   638→            break;
   639→          }
   640→          
   641→          concepto += ' ' + lineaSiguiente.trim();
   642→          j++;
   643→        }
   644→      }
   645→      
   646→      if (montoEncontrado === null || Math.abs(montoEncontrado) < 0.5) { i++; continue; }
   647→      
   648→      // Limpiar concepto: quitar folio al inicio (7 dígitos) y montos
   649→      concepto = concepto.replace(/^\s*\d{7}\s*/, '');
   650→      concepto = quitarMontos(concepto).replace(/\s+/g, ' ').trim().slice(0, 300);
   651→      if (!concepto) concepto = 'Movimiento bancario';
   652→      
   653→      // Determinar signo
   654→      let montoFinal = montoEncontrado;
   655→      if (saldoEncontrado !== null && saldoAnterior !== null) {
   656→        const diferencia = saldoEncontrado - saldoAnterior;
   657→        if (Math.abs(Math.abs(diferencia) - montoEncontrado) < montoEncontrado * 0.02) {
   658→          montoFinal = diferencia;
   659→        } else if (diferencia < 0) {
   660→          montoFinal = -Math.abs(montoEncontrado);
   661→        } else {
   662→          montoFinal = Math.abs(montoEncontrado);
   663→        }
   664→      } else {
   665→        // Fallback por keywords
   666→        const conceptoUpper = concepto.toUpperCase();
   667→        const esRetiro = ['COMPRA', 'PAGO', 'RETIRO', 'CARGO', 'TRASPASO', 'COMISION', 'COMISIÓN', 
   668→                          'TRANSFERENCIA', 'INTERESES EXENTO', 'PAGO DE CAPITAL', 'PAGO DE CREDITO', 
   669→                          'ADMINISTRACION', 'CGO', 'CARGO CAPITAL', 'CARGO POR', 'CGO INTERESES',
   670→                          'I V A POR COMISION', 'PAGO PRIMA SEGURO', 'I.V.A.'].some(k => conceptoUpper.includes(k));
   671→        const esDeposito = ['DISPOSICION', 'RECIBIDO', 'DEPOSITO DE CUENTA', 'DEV. DEPOSITO', 
   672→                            'DEVOLUCION', 'ABONO', 'APORT LINEA CAPTURA'].some(k => conceptoUpper.includes(k));
   673→        if (esRetiro) montoFinal = -Math.abs(montoEncontrado);
   674→        else if (esDeposito) montoFinal = Math.abs(montoEncontrado);
   675→      }
   676→      
   677→      if (saldoEncontrado !== null) saldoAnterior = saldoEncontrado;
   678→      
   679→      movimientos.push({
   680→        fecha,
   681→        concepto,
   682→        monto: montoFinal,
   683→        ...(seccion.tipo === 'inversion' ? { esInversion: true } : {}),
   684→      } as any);
   685→      
   686→      i = lineaMontoIdx > 0 ? lineaMontoIdx + 1 : i + 1;
   687→    }
   688→    
   689→    return movimientos;
   690→  }
   691→
   692→  // ===== FASE 3: PROCESAR TODAS LAS SECCIONES =====
   693→  const todosMovimientos: MovimientoImportado[] = [];
   694→  const seccionesDetectadas: Array<{ tipo: string; cuentaNumero: string; count: number; saldoInicial: number | null }> = [];
   695→  
   696→  for (const seccion of secciones) {
   697→    const movs = parsearSeccion(seccion);
   698→    todosMovimientos.push(...movs);
   699→    seccionesDetectadas.push({
   700→      tipo: seccion.tipo,
   701→      cuentaNumero: seccion.cuentaNum,
   702→      count: movs.length,
   703→      saldoInicial: seccion.saldoInicial,
   704→    });
   705→  }
   706→
   707→  return { movimientos: todosMovimientos, seccionesDetectadas };
   708→}

   710→// Extrae montos numéricos de una línea (formato: 80,000.00 o $1,234.56)
   711→function extraerMontos(linea: string): number[] {
   712→  const montos: number[] = [];
   713→  const regex = /-?\$?\s?[\d,]+\.\d{2}/g;
   714→  let match;
   715→  while ((match = regex.exec(linea)) !== null) {
   716→    const valor = parseFloat(match[0].replace(/[$,\s]/g, ''));
   717→    if (!isNaN(valor) && Math.abs(valor) > 0.5) {
   718→      montos.push(valor);
   719→    }
   720→  }
   721→  return montos;
   722→}
   723→
   724→// Quita los montos de un texto para dejar solo la descripción
   725→function quitarMontos(texto: string): string {
   726→  return texto.replace(/-?\$?\s?[\d,]+\.\d{2}/g, '').trim();
   727→}
