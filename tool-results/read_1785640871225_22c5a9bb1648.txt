     5→/**
     6→ * GET /api/export/concentrado?empresaId=xxx&anio=2026
     7→ *
     8→ * Genera un Excel que replica EXACTAMENTE la estructura del concentrado
     9→ * mensual del usuario:
    10→ *
    11→ * Hojas:
    12→ *   - Ene, Feb, Mar, ..., Dic  (una por mes con CFDIs del mes)
    13→ *   - Concentrado (tabla pivote con totales Emitidas/Recibidas/Nómina por mes)
    14→ *   - NOMINA (CFDIs tipo Nómina separados)
    15→ *
    16→ * Cada hoja mensual tiene las columnas:
    17→ *   XML | Rfc Emisor | Nombre Emisor | LugarExp | Régimen Fiscal |
    18→ *   Rfc Receptor | Nombre Receptor | Tipo | Serie | Folio | Fecha |
    19→ *   Sub Total | Descuento | Total impuesto Trasladado | Nombre Impuesto |
    20→ *   Total impuesto Retenido | Nombre Impuesto | Total | UUID |
    21→ *   Método de Pago | Forma de Pago | Moneda | Tipo de Cambio |
    22→ *   Versión | Uso CFDI | Régimen Fiscal | Estado | Estatus |
    23→ *
    24→ * Filtros:
    25→ *   - Excluye CFDIs cancelados
    26→ *   - Excluye CFDIs tipo "P" (Pago) — van en su propio reporte
    27→ *   - Incluye tipo "I" (Ingreso) y "E" (Nota de crédito)
    28→ *   - Nómina (tipo "N") va en hoja separada
    29→ */

    65→export async function GET(req: NextRequest) {
    66→  try {
    67→    const { searchParams } = new URL(req.url);
    68→    const hoy = new Date();

    70→    const empresaId = searchParams.get('empresaId') || undefined;
    71→

    73→    const empresa = empresaId ? await db.empresa.findUnique({ where: { id: empresaId } }) : null;
    74→

    78→

    83→        estado: { not: 'cancelada' },

    87→    });
    88→
    89→    // ===== Obtener nómina del año =====
    90→    const nominas = await db.reciboNomina.findMany({
    91→      where: {
    92→        ...(empresaId ? { empresaId } : {}),
    93→        fecha: { gte: inicioAnio, lte: finAnio },

   102→    wb.created = new Date();

   110→      nomina: { total: number; count: number };
   111→    }> = [];
   112→
   113→    for (let mes = 0; mes < 12; mes++) {
   114→      const facturasMes = facturas.filter(f => new Date(f.fecha).getMonth() === mes);

   116→
   117→      // ===== SEPARAR INGRESOS (I) DE NOTAS DE CRÉDITO (E) =====

   119→      // Las Notas de Crédito (E) se RESTAN de los totales (son devoluciones)

   123→      const notasCreditoRecibidas = facturasMes.filter(f => f.direccion === 'recibida' && f.tipoComprobante === 'E');
   124→
   125→      // ===== TOTALES: Ingresos - Notas de Crédito =====

   127→        subtotal: ingresosEmitidos.reduce((s, f) => s + f.subtotal, 0) - notasCreditoEmitidas.reduce((s, f) => s + f.subtotal, 0),
   128→        descuento: ingresosEmitidos.reduce((s, f) => s + (f.descuento || 0), 0) + notasCreditoEmitidas.reduce((s, f) => s + (f.descuento || 0), 0),
   129→        impuesto: ingresosEmitidos.reduce((s, f) => s + f.totalImpuestos, 0) - notasCreditoEmitidas.reduce((s, f) => s + f.totalImpuestos, 0),
   130→        retenido: ingresosEmitidos.reduce((s, f) => s + (f.impuestoRetenido || 0), 0) - notasCreditoEmitidas.reduce((s, f) => s + (f.impuestoRetenido || 0), 0),
   131→        total: ingresosEmitidos.reduce((s, f) => s + f.total, 0) - notasCreditoEmitidas.reduce((s, f) => s + f.total, 0),
   132→        count: ingresosEmitidos.length + notasCreditoEmitidas.length,
   133→      };
   134→      const totalesRecibidas = {
   135→        subtotal: ingresosRecibidos.reduce((s, f) => s + f.subtotal, 0) - notasCreditoRecibidas.reduce((s, f) => s + f.subtotal, 0),
   136→        descuento: ingresosRecibidos.reduce((s, f) => s + (f.descuento || 0), 0) + notasCreditoRecibidas.reduce((s, f) => s + (f.descuento || 0), 0),
   137→        impuesto: ingresosRecibidos.reduce((s, f) => s + f.totalImpuestos, 0) - notasCreditoRecibidas.reduce((s, f) => s + f.totalImpuestos, 0),
   138→        retenido: ingresosRecibidos.reduce((s, f) => s + (f.impuestoRetenido || 0), 0) - notasCreditoRecibidas.reduce((s, f) => s + (f.impuestoRetenido || 0), 0),
   139→        total: ingresosRecibidos.reduce((s, f) => s + f.total, 0) - notasCreditoRecibidas.reduce((s, f) => s + f.total, 0),
   140→        count: ingresosRecibidos.length + notasCreditoRecibidas.length,

   165→
   166→      // Estilo header

   168→      ws.getRow(1).fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: COLORES.header } };

   170→      ws.getRow(1).height = 30;
   171→
   172→      // Mapear facturas del mes a filas

   183→          regimen_fiscal: empresa?.regimenFiscal || '',

   199→          forma_de_pago: f.formaPago || '',
   200→          moneda: f.moneda || 'MXN',
   201→          tipo_de_cambio: '',
   202→          version: '4.0',
   203→          uso_cfdi: '',
   204→          regimen_fiscal_2: '',
   205→          estado: f.estado || 'timbrada',
   206→          estatus: 'Vigente',

   212→      for (const f of notasCreditoEmitidas) {

   216→          nombre_emisor: f.emisorNombre || empresa?.nombre || '',

   225→          sub_total: -f.subtotal,  // NEGATIVO
   226→          descuento: f.descuento || 0,
   227→          total_impuesto_trasladado: -f.totalImpuestos,  // NEGATIVO
   228→          nombre_impuesto: '002 - IVA',
   229→          total_impuesto_retenido: -(f.impuestoRetenido || 0),  // NEGATIVO
   230→          nombre_impuesto_2: f.impuestoRetenido ? '001 - ISR' : '',

   232→          uuid: f.uuid || '',
   233→          metodo_de_pago: f.metodoPago || '',

   248→        filasMes.push({
   249→          xml: f.uuid || '',
   250→          rfc_emisor: f.emisorRfc || '',
   251→          nombre_emisor: f.emisorNombre || '',
   252→          lugarexp: '',
   253→          regimen_fiscal: '',
   254→          rfc_receptor: f.receptorRfc || empresa?.rfc || '',
   255→          nombre_receptor: f.receptorNombre || empresa?.nombre || '',
   256→          tipo: 'egreso',
   257→          serie: f.serie || '',
   258→          folio: f.folio || '',

   260→          sub_total: f.subtotal,
   261→          descuento: f.descuento || 0,
   262→          total_impuesto_trasladado: f.totalImpuestos,
   263→          nombre_impuesto: '002 - IVA',
   264→          total_impuesto_retenido: f.impuestoRetenido || 0,
   265→          nombre_impuesto_2: f.impuestoRetenido ? '001 - ISR' : '',
   266→          total: f.total,
   267→          uuid: f.uuid || '',
   268→          metodo_de_pago: f.metodoPago || '',
   269→          forma_de_pago: f.formaPago || '',
   270→          moneda: f.moneda || 'MXN',
   271→          tipo_de_cambio: '',
   272→          version: '4.0',
   273→          uso_cfdi: '',
   274→          regimen_fiscal_2: '',
   275→          estado: f.estado || 'timbrada',

   291→          tipo: 'nota de crédito',

   293→          folio: f.folio || '',

   295→          sub_total: -f.subtotal,  // NEGATIVO
